# Extra packages the alignment bake-off needs on top of the `wmcompare` env.
# Everything else (numpy, scipy, librosa, torch, transformers, pydub, ffmpeg,
# matplotlib) is already there.
#
# Decision: install into `wmcompare` directly rather than a separate env.
#
# BEFORE installing, snapshot the versions that matter so a bad resolve can be
# undone:
#     conda activate wmcompare
#     pip freeze > $HOME/wmcompare_freeze_before_align.txt
#     pip list | grep -Ei "numpy|scipy|librosa|torch|numba"
#
# See what pip WOULD change before letting it:
#     pip install --dry-run -r requirements_extra.txt
#
# If it proposes touching numpy, scipy, numba or torch, stop and install with
#     pip install --no-deps -r requirements_extra.txt
# then add only the genuinely missing leaf deps by hand.
#
# AFTER installing, confirm nothing broke:
#     python -c "from aware.utils.models import load; load(name='AWARE(20bps)'); print('aware ok')"
#     python -c "import audioseal; print('audioseal ok')"
#
# Rollback if needed:
#     pip install -r $HOME/wmcompare_freeze_before_align.txt

# KNOWN CONFLICT (observed 2026-08-05 on Great Lakes):
#   audio-offset-finder requires librosa >= 0.10.2
#   aware               requires librosa == 0.9.2
# Installing both upgrades librosa 0.9.2 -> 0.11.0 and breaks AWARE's pin. It
# also pulls matplotlib 3.7.2 -> 3.8.2, tqdm 4.66.1 -> 4.66.2, and DOWNGRADES
# setuptools 70.2.0 -> 59.6.0.
#
# audalign alone does NOT need the librosa upgrade -- only audio-offset-finder
# does. So if you must stay in `wmcompare`, install audalign and skip aof:
#     pip install audalign
# You lose one of six methods, and it is the weakest one: aof is MFCC
# cross-correlation, which gcc_phat already covers with better resolution.
